# 46. Работа с почтой

Данный раздел описывает систему почтовых сервисов в платформе DST Platform. Система обеспечивает надёжную отправку транзакционных писем через внешние SMTP-серверы, интеграцию с компонентами платформы и комплексную обработку ошибок.

---

## 46.1. Обзор системы

Платформа поддерживает два основных подхода к отправке почты:

| Подход | Реализация | Назначение | Статус |
|--------|------------|------------|--------|
| **Системный сервис** | `cmsMailer` | Интеграция с ядром для транзакционных писем | ✅ Рекомендуется |
| **Автономный обработчик** | `contact_process.php` | Простые формы обратной связи | ✅ Дополнительно |

### Ключевые характеристики

| Характеристика | Реализация |
|----------------|------------|
| **Библиотека** | PHPMailer (через Composer) |
| **SMTP-сервер** | Яндекс, Gmail, Mail.ru и др. |
| **Шифрование** | TLS/SSL |
| **Формат писем** | HTML с текстовой альтернативой |
| **Вложения** | Поддержка файлов до 64 МБ |
| **Санитизация** | `strip_tags()` + `htmlspecialchars()` |
| **Контекст выполнения** | Только через системный сервис |

### Интеграция с ядром

```php
// /system/core/mailer.php
class cmsMailer {
    private $mailer;
    private $config;
    
    public function __construct() {
        $this->config = cmsConfig::getInstance();
        $this->initMailer();
    }
    
    private function initMailer() {
        $this->mailer = new PHPMailer(true);
        
        // Настройка из конфигурации
        $this->mailer->isSMTP();
        $this->mailer->Host = $this->config->get('smtp_host', 'smtp.yandex.ru');
        $this->mailer->SMTPAuth = true;
        $this->mailer->Username = $this->config->get('smtp_username');
        $this->mailer->Password = $this->config->get('smtp_password');
        $this->mailer->SMTPSecure = $this->config->get('smtp_secure', PHPMailer::ENCRYPTION_STARTTLS);
        $this->mailer->Port = $this->config->get('smtp_port', 587);
        $this->mailer->CharSet = 'UTF-8';
    }
    
    public function setTo($email, $name = '') {
        $this->mailer->addAddress($email, $name);
        return $this;
    }
    
    public function setSubject($subject) {
        $this->mailer->Subject = $subject;
        return $this;
    }
    
    public function setBody($body, $is_html = true) {
        if ($is_html) {
            $this->mailer->isHTML(true);
            $this->mailer->Body = $body;
            $this->mailer->AltBody = strip_tags($body);
        } else {
            $this->mailer->isHTML(false);
            $this->mailer->Body = $body;
        }
        return $this;
    }
    
    public function addAttachment($path, $name = '') {
        $this->mailer->addAttachment($path, $name);
        return $this;
    }
    
    public function send() {
        $start_time = microtime(true);
        
        try {
            $result = $this->mailer->send();
            
            // Логирование успешной отправки
            $duration = microtime(true) - $start_time;
            cmsDebugging::log('email_sent', [
                'to' => $this->mailer->getToAddresses(),
                'subject' => $this->mailer->Subject,
                'duration' => round($duration, 4),
                'size' => strlen($this->mailer->Body),
                'timestamp' => date('Y-m-d H:i:s')
            ]);
            
            return $result;
            
        } catch (Exception $e) {
            // Логирование ошибки
            cmsDebugging::log('email_error', [
                'to' => $this->mailer->getToAddresses(),
                'subject' => $this->mailer->Subject,
                'error' => $this->mailer->ErrorInfo,
                'exception' => $e->getMessage(),
                'timestamp' => date('Y-m-d H:i:s')
            ]);
            
            return false;
        }
    }
}
```

```php
// /system/controllers/auth/mailer.php
class authMailer {
    
    /**
     * Отправка письма подтверждения регистрации
     */
    public function sendRegistrationConfirmation($user, $confirm_token) {
        $mailer = cmsMailer::getInstance();
        
        // Генерация ссылки подтверждения
        $confirm_url = cmsConfig::get('host') . '/auth/confirm/' . $confirm_token;
        
        // Рендеринг шаблона
        $template = $this->renderTemplate('registration_confirmation', [
            'user' => $user,
            'confirm_url' => $confirm_url,
            'site_name' => cmsConfig::get('sitename')
        ]);
        
        $mailer->setTo($user['email'], $user['nickname'])
            ->setSubject('Подтверждение регистрации на ' . cmsConfig::get('sitename'))
            ->setBody($template, true)
            ->send();
    }
    
    /**
     * Рендеринг шаблона письма
     */
    private function renderTemplate($template_name, $data) {
        $template_path = PATH . '/templates/default/mail/' . $template_name . '.tpl.php';
        
        if (!file_exists($template_path)) {
            throw new Exception("Шаблон письма не найден: {$template_name}");
        }
        
        extract($data);
        ob_start();
        include $template_path;
        return ob_get_clean();
    }
}
```

### Массовая отправка писем

```php
/**
 * Массовая отправка уведомлений
 */
public function sendBulkNotifications($users, $subject, $template_name, $data) {
    $mailer = cmsMailer::getInstance();
    $sent_count = 0;
    $failed_count = 0;
    
    foreach ($users as $user) {
        try {
            // Подготовка данных для шаблона
            $template_data = array_merge($data, [
                'user' => $user
            ]);
            
            // Рендеринг шаблона
            $body = $this->renderTemplate($template_name, $template_data);
            
            // Отправка письма
            $mailer->setTo($user['email'], $user['nickname'])
                ->setSubject($subject)
                ->setBody($body, true)
                ->send();
            
            $sent_count++;
            
            // Пауза между отправками для предотвращения блокировки
            usleep(100000); // 100 мс
            
        } catch (Exception $e) {
            error_log("Ошибка отправки письма {$user['email']}: " . $e->getMessage());
            $failed_count++;
        }
    }
    
    return [
        'sent' => $sent_count,
        'failed' => $failed_count,
        'total' => count($users)
    ];
}
```

## 46.6. Безопасность

### Санитизация входных данных

```php
/**
 * Безопасная обработка данных для письма
 */
class EmailSecurity {
    
    /**
     * Санитизация текста
     */
    public static function sanitizeText($text) {
        $text = trim($text);
        $text = htmlspecialchars($text, ENT_QUOTES, 'UTF-8');
        return strip_tags($text);
    }
    
    /**
     * Санитизация заголовка
     */
    public static function sanitizeSubject($subject) {
        // Удаление потенциально опасных символов
        $subject = preg_replace('/[\r\n\t]/', ' ', $subject);
        return htmlspecialchars($subject, ENT_QUOTES, 'UTF-8');
    }
    
    /**
     * Валидация email-адреса
     */
    public static function validateEmail($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }
    
    /**
     * Проверка на спам
     */
    public static function isSpam($content) {
        $spam_keywords = ['viagra', 'casino', 'xxx', 'free money', 'казино', 'онлайн казино'];
        $content_lower = mb_strtolower($content);
        
        foreach ($spam_keywords as $keyword) {
            if (mb_strpos($content_lower, $keyword) !== false) {
                return true;
            }
        }
        
        return false;
    }
}

// Использование
$email = EmailSecurity::sanitizeText($_POST['email']);
$subject = EmailSecurity::sanitizeSubject($_POST['subject']);
$body = EmailSecurity::sanitizeText($_POST['body']);

if (!EmailSecurity::validateEmail($email)) {
    die('Неверный формат email');
}

if (EmailSecurity::isSpam($body)) {
    die('Обнаружен спам');
}
```

```php
class EmailHeaderSecurity {
    
    public static function sanitizeHeader($header) {
        // Удаление символов перевода строки
        $header = str_replace(["\r", "\n", "%0A", "%0D"], '', $header);
```

### Ограничение размера вложений

```php
/**
 * Проверка размера вложений
 */
class EmailAttachmentSecurity {
    
    const MAX_ATTACHMENT_SIZE = 10 * 1024 * 1024; // 10 МБ
    const MAX_TOTAL_SIZE = 25 * 1024 * 1024; // 25 МБ
    const ALLOWED_EXTENSIONS = [
        'jpg', 'jpeg', 'png', 'gif', 'pdf', 'doc', 'docx', 'xls', 'xlsx'
    ];
    
    public static function validateAttachment($file) {
        // Проверка размера
        if ($file['size'] > self::MAX_ATTACHMENT_SIZE) {
            return [
                'valid' => false,
                'error' => 'Файл слишком большой. Максимум ' . (self::MAX_ATTACHMENT_SIZE / 1024 / 1024) . ' МБ'
            ];
        }
        
        // Проверка расширения
        $extension = mb_strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        if (!in_array($extension, self::ALLOWED_EXTENSIONS)) {
            return [
                'valid' => false,
                'error' => 'Недопустимый формат файла'
            ];
        }
        
        // Проверка типа файла
        $finfo = finfo_open(FILEINFO_MIME_TYPE);
        $mime_type = finfo_file($finfo, $file['tmp_name']);
        finfo_close($finfo);
        
        $allowed_mimes = [
            'image/jpeg', 'image/png', 'image/gif', 'application/pdf',
            'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
            'application/vnd.ms-excel', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
        ];
        
        if (!in_array($mime_type, $allowed_mimes)) {
            return [
                'valid' => false,
                'error' => 'Недопустимый тип файла'
            ];
        }
        
        return ['valid' => true];
    }
    
    public static function validateTotalSize($attachments) {
        $total_size = 0;
        
        foreach ($attachments as $file) {
            $total_size += $file['size'];
        }
        
        if ($total_size > self::MAX_TOTAL_SIZE) {
            return [
                'valid' => false,
                'error' => 'Общий размер вложений превышает ' . (self::MAX_TOTAL_SIZE / 1024 / 1024) . ' МБ'
            ];
        }
        
        return ['valid' => true];
    }
}
```

```php
/**
 * Расширенная обработка ошибок отправки
 */
class EmailErrorHandler {
    
    public static function handleError($mailer, $error_info) {
        $error_code = self::parseErrorCode($error_info);
        
        switch ($error_code) {
            case 'authentication_failed':
                error_log('Ошибка аутентификации SMTP');
                self::sendAdminAlert('Ошибка SMTP аутентификации', $error_info);
                break;
                
            case 'connection_refused':
                error_log('Не удалось подключиться к SMTP-серверу');
                self::sendAdminAlert('Ошибка подключения к SMTP', $error_info);
                break;
                
            case 'recipient_rejected':
                error_log('Получатель отклонён сервером');
                // Можно добавить в чёрный список
                break;
                
            case 'quota_exceeded':
                error_log('Превышена квота отправки');
                self::sendAdminAlert('Превышена квота SMTP', $error_info);
                break;
                
            default:
                error_log('Неизвестная ошибка отправки: ' . $error_info);
                break;
        }
    }
    
    private static function parseErrorCode($error_info) {
        $error_info_lower = mb_strtolower($error_info);
        
        if (mb_strpos($error_info_lower, 'authentication') !== false) {
            return 'authentication_failed';
        }
        
        if (mb_strpos($error_info_lower, 'connection') !== false || 
            mb_strpos($error_info_lower, 'connect') !== false) {
            return 'connection_refused';
        }
        
        if (mb_strpos($error_info_lower, 'recipient') !== false || 
            mb_strpos($error_info_lower, 'relay') !== false) {
            return 'recipient_rejected';
        }
        
        if (mb_strpos($error_info_lower, 'quota') !== false || 
            mb_strpos($error_info_lower, 'limit') !== false) {
            return 'quota_exceeded';
        }
        
        return 'unknown';
    }
    
    private static function sendAdminAlert($subject, $error_info) {
        $admin_email = cmsConfig::get('admin_email');
        
        $mailer = cmsMailer::getInstance();
        $mailer->setTo($admin_email)
            ->setSubject('[ALERT] ' . $subject)
            ->setBody("Ошибка отправки почты:\n\n" . $error_info, false)
            ->send();
    }
}

// Использование
try {
    $mailer->send();
} catch (Exception $e) {
    EmailErrorHandler::handleError($mailer, $mailer->ErrorInfo);
    throw $e; // Переброс исключения для обработки на уровне приложения
}
```

```php
/**
 * Логирование отправки писем
 */
class EmailLogger {
    
    /**
     * Запись в лог отправки письма
     */
    public static function logSend($email_data) {
        $log_entry = [
            'timestamp' => date('Y-m-d H:i:s'),
            'to' => $email_data['to'],
            'subject' => $email_data['subject'],
            'type' => $email_data['type'] ?? 'unknown',
            'status' => $email_data['status'],
            'duration' => $email_data['duration'] ?? 0,
            'size' => $email_data['size'] ?? 0
        ];
        
        // Запись в файл
        $log_file = cmsConfig::get('cache_path') . 'email_log.json';
        $logs = file_exists($log_file) ? json_decode(file_get_contents($log_file), true) : [];
        
        array_unshift($logs, $log_entry);
        
        // Ограничение размера лога (последние 1000 записей)
        $logs = array_slice($logs, 0, 1000);
        
        file_put_contents($log_file, json_encode($logs, JSON_PRETTY_PRINT));
    }
    
    /**
     * Получение статистики отправки
     */
    public static function getStatistics($days = 30) {
        $log_file = cmsConfig::get('cache_path') . 'email_log.json';
        
        if (!file_exists($log_file)) {
            return [
                'total_sent' => 0,
                'failed' => 0,
                'success_rate' => 0,
                'by_type' => []
            ];
        }
        
        $logs = json_decode(file_get_contents($log_file), true);
        $cutoff_date = strtotime("-{$days} days");
        
        $stats = [
            'total_sent' => 0,
            'failed' => 0,
            'success_rate' => 0,
            'by_type' => []
        ];
        
        foreach ($logs as $log) {
            $log_time = strtotime($log['timestamp']);
            
            if ($log_time < $cutoff_date) {
                continue;
            }
            
            $stats['total_sent']++;
            
            if ($log['status'] !== 'sent') {
                $stats['failed']++;
            }
            
            $type = $log['type'];
            if (!isset($stats['by_type'][$type])) {
                $stats['by_type'][$type] = ['sent' => 0, 'failed' => 0];
            }
            
            if ($log['status'] === 'sent') {
                $stats['by_type'][$type]['sent']++;
            } else {
                $stats['by_type'][$type]['failed']++;
            }
        }
        
        $stats['success_rate'] = $stats['total_sent'] > 0 
            ? round(($stats['total_sent'] - $stats['failed']) / $stats['total_sent'] * 100, 2)
            : 0;
        
        return $stats;
    }
}

// Использование в отправке
$mailer = cmsMailer::getInstance();
$start_time = microtime(true);

try {
    $result = $mailer->setTo($email)
        ->setSubject($subject)
        ->setBody($body, true)
        ->send();
    
    $duration = microtime(true) - $start_time;
    
    EmailLogger::logSend([
        'to' => $email,
        'subject' => $subject,
        'type' => 'registration',
        'status' => $result ? 'sent' : 'failed',
        'duration' => $duration,
        'size' => strlen($body)
    ]);
    
} catch (Exception $e) {
    EmailLogger::logSend([
        'to' => $email,
        'subject' => $subject,
        'type' => 'registration',
        'status' => 'error',
        'error' => $e->getMessage()
    ]);
}
```

### Проблема: "Превышена квота отправки"

**Симптомы:**
```

```

**Решение:**
```php
// 1. Ограничение частоты отправки
class EmailRateLimiter {
    private $cache;
    
    public function __construct($cache) {
        $this->cache = $cache;
    }
    
    public function checkRateLimit($identifier, $limit = 100, $window = 3600) {
        $cache_key = 'email_rate_limit:' . $identifier;
        $attempts = $this->cache->get($cache_key) ?? 0;
        
        if ($attempts >= $limit) {
            return false;
        }
        
        $this->cache->set($cache_key, $attempts + 1, $window);
        return true;
    }
}

// Использование
$rate_limiter = new EmailRateLimiter(cmsCache::getInstance());

if (!$rate_limiter->checkRateLimit($user_email)) {
    error_log("Rate limit exceeded for {$user_email}");
    return false;
}
```
