# 43. Пример полного компонента (от модели до шаблона)

Данный раздел демонстрирует создание полноценного компонента DST Platform на примере блога. Рассматривается полная структура от модели данных до шаблонов представления, включая административную панель, права доступа и интеграцию с другими компонентами.

---

## 43.1. Структура компонента

Полный компонент включает следующие элементы:

```
/system/controllers/blog/
├── frontend.php              # Фронтенд-контроллер (публичная часть)
├── model.php                 # Модель данных
├── manifest.php              # Метаданные компонента
├── hooks.php                 # Обработчики событий (опционально)
├── backend/
│   ├── controller.php        # Бэкенд-контроллер (админка)
│   └── model.php             # Модель админки (расширяет основную модель)
└── widgets/
    └── latest.php            # Виджет "Последние посты"

/templates/default/controllers/blog/
├── index.tpl.php             # Список постов
├── view.tpl.php              # Детальная страница поста
├── form.tpl.php              # Форма создания/редактирования
├── backend/
│   ├── index.tpl.php         # Список в админке
│   └── form.tpl.php          # Форма в админке
└── styles.css                # Стили компонента
```

## 43.3. Модель данных (model.php)

```php
<?php
/**
 * model.php
 * Модель данных блога
 */

class modelBlog extends cmsModel {
    
    /**
     * Получение всех постов с фильтрацией
     * @param array $filter Параметры фильтрации
     * @return array
     */
    public function getPosts($filter = []) {
        $this->useCache(true);
        
        // Базовый запрос
        $this->select('blog_posts', 'p');
        $this->select('users', 'u', 'u.id = p.user_id');
        
        // Фильтрация по статусу публикации
        if (empty($filter['show_all'])) {
            $this->filterEqual('p.is_published', 1);
            $this->filterEqual('p.is_deleted', 0);
            $this->filterIsNull('p.date_pub', false);
            $this->filterLte('p.date_pub', date('Y-m-d H:i:s'));
        }
        
        // Фильтрация по категории
        if (!empty($filter['category_id'])) {
            $this->filterEqual('p.category_id', $filter['category_id']);
        }
        
        // Фильтрация по автору
        if (!empty($filter['user_id'])) {
            $this->filterEqual('p.user_id', $filter['user_id']);
        }
        
        // Фильтрация по тегам
        if (!empty($filter['tag'])) {
            $this->filterLike('p.tags', '%' . $filter['tag'] . '%');
        }
        
        // Сортировка
        $order_by = !empty($filter['order_by']) ? $filter['order_by'] : 'p.date_pub';
        $order_to = !empty($filter['order_to']) ? $filter['order_to'] : 'DESC';
        $this->orderBy($order_by, $order_to);
        
        // Пагинация
        if (!empty($filter['page'])) {
            $per_page = !empty($filter['per_page']) ? $filter['per_page'] : 10;
            $this->limitPage($filter['page'], $per_page);
        }
        
        // Выполнение запроса
        $posts = $this->get();
        
        // Обогащение данными
        foreach ($posts as &$post) {
            $post = $this->preparePost($post);
        }
        
        return $posts;
    }
    
    /**
     * Получение одного поста по ID
     * @param int $id ID поста
     * @param bool $inc_views Увеличить счётчик просмотров
     * @return array|null
     */
    public function getPost($id, $inc_views = true) {
        $post = $this->getItemById('blog_posts', $id);
        
        if (!$post) {
            return null;
        }
        
        // Увеличение счётчика просмотров
        if ($inc_views && $post['is_published']) {
            $this->increment('blog_posts', $id, 'views_count');
        }
        
        return $this->preparePost($post);
    }
    
    /**
     * Подготовка данных поста
     * @param array $post Данные поста
     * @return array
     */
    private function preparePost($post) {
        // Получение автора
        $post['author'] = $this->getItemById('users', $post['user_id']);
        
        // Получение категории
        if ($post['category_id']) {
            $post['category'] = $this->getItemById('blog_categories', $post['category_id']);
        }
        
        // Парсинг тегов
        if (!empty($post['tags'])) {
            $post['tags_array'] = array_filter(array_map('trim', explode(',', $post['tags'])));
        } else {
            $post['tags_array'] = [];
        }
        
        // Форматирование дат
        $post['date_pub_formatted'] = !empty($post['date_pub']) ? 
            date('d.m.Y H:i', strtotime($post['date_pub'])) : '';
        $post['date_created_formatted'] = date('d.m.Y H:i', strtotime($post['date_created']));
        
        // Генерация URL
        $post['url'] = href_to('blog', 'view', [$post['id'], $post['slug']]);
        
        return $post;
    }
    
    /**
     * Получение категорий
     * @param int $parent_id ID родительской категории
     * @return array
     */
    public function getCategories($parent_id = 0) {
        return $this->filterEqual('parent_id', $parent_id)
            ->orderBy('sort_order', 'ASC')
            ->orderBy('title', 'ASC')
            ->get('blog_categories');
    }
    
    /**
     * Получение категории по слагу
     * @param string $slug Слаг категории
     * @return array|null
     */
    public function getCategoryBySlug($slug) {
        return $this->filterEqual('slug', $slug)
            ->getItem('blog_categories');
    }
    
    /**
     * Сохранение поста
     * @param array $data Данные поста
     * @param int|null $id ID для обновления (null для создания)
     * @return int ID поста
     */
    public function savePost($data, $id = null) {
        // Генерация слага
        if (empty($data['slug'])) {
            $data['slug'] = $this->generateSlug($data['title']);
        }
        
        // Установка даты публикации
        if (empty($data['date_pub']) && !empty($data['is_published'])) {
            $data['date_pub'] = date('Y-m-d H:i:s');
        }
        
        // Установка даты создания/обновления
        if ($id) {
            $data['date_updated'] = date('Y-m-d H:i:s');
        } else {
            $data['date_created'] = date('Y-m-d H:i:s');
        }
        
        // Сохранение в БД
        if ($id) {
            $this->update('blog_posts', $id, $data);
            $post_id = $id;
        } else {
            $post_id = $this->insert('blog_posts', $data);
        }
        
        // Очистка кэша
        $this->cleanCache();
        
        // Вызов хуков
        $item = $this->getItemById('blog_posts', $post_id);
        $event = $id ? 'blog_after_update' : 'blog_after_add';
        cmsEventsManager::hook($event, $item);
        
        return $post_id;
    }
    
    /**
     * Удаление поста (мягкое удаление)
     * @param int $id ID поста
     * @return bool
     */
    public function deletePost($id) {
        $result = $this->update('blog_posts', $id, [
            'is_deleted' => 1,
            'date_updated' => date('Y-m-d H:i:s')
        ]);
        
        if ($result) {
            $this->cleanCache();
            $item = $this->getItemById('blog_posts', $id);
            cmsEventsManager::hook('blog_after_delete', $item);
        }
        
        return $result;
    }
    
    /**
     * Восстановление поста
     * @param int $id ID поста
     * @return bool
     */
    public function restorePost($id) {
        return $this->update('blog_posts', $id, [
            'is_deleted' => 0,
            'date_updated' => date('Y-m-d H:i:s')
        ]);
    }
    
    /**
     * Генерация уникального слага
     * @param string $title Заголовок
     * @param int|null $exclude_id ID поста для исключения
     * @return string
     */
    public function generateSlug($title, $exclude_id = null) {
        $slug = translit($title, true);
        $slug = mb_strtolower($slug);
        $slug = preg_replace('/[^a-z0-9\-_]/', '-', $slug);
        $slug = preg_replace('/\-+/', '-', $slug);
        $slug = trim($slug, '-');
        
        // Проверка уникальности
        $counter = 0;
        $original_slug = $slug;
        
        while ($this->isSlugExists($slug, $exclude_id)) {
            $counter++;
            $slug = $original_slug . '-' . $counter;
        }
        
        return $slug;
    }
    
    /**
     * Проверка существования слага
     * @param string $slug Слаг
     * @param int|null $exclude_id ID для исключения
     * @return bool
     */
    private function isSlugExists($slug, $exclude_id = null) {
        $this->filterEqual('slug', $slug);
        
        if ($exclude_id) {
            $this->filterNotEqual('id', $exclude_id);
        }
        
        return (bool)$this->getItem('blog_posts');
    }
    
    /**
     * Очистка кэша
     */
    private function cleanCache() {
        cmsCache::getInstance()->clean('blog_posts');
        cmsCache::getInstance()->clean('blog_categories');
    }
    
    /**
     * Получение популярных постов
     * @param int $limit Количество постов
     * @return array
     */
    public function getPopularPosts($limit = 5) {
        return $this->filterEqual('is_published', 1)
            ->filterEqual('is_deleted', 0)
            ->filterLte('date_pub', date('Y-m-d H:i:s'))
            ->orderBy('views_count', 'DESC')
            ->limit($limit)
            ->get('blog_posts');
    }
    
    /**
     * Получение последних постов
     * @param int $limit Количество постов
     * @return array
     */
    public function getLatestPosts($limit = 5) {
        return $this->filterEqual('is_published', 1)
            ->filterEqual('is_deleted', 0)
            ->filterLte('date_pub', date('Y-m-d H:i:s'))
            ->orderBy('date_pub', 'DESC')
            ->limit($limit)
            ->get('blog_posts');
    }
    
    /**
     * Получение постов по тегу
     * @param string $tag Тег
     * @param int $limit Количество постов
     * @return array
     */
    public function getPostsByTag($tag, $limit = 10) {
        return $this->filterEqual('is_published', 1)
            ->filterEqual('is_deleted', 0)
            ->filterLte('date_pub', date('Y-m-d H:i:s'))
            ->filterLike('tags', '%' . $tag . '%')
            ->orderBy('date_pub', 'DESC')
            ->limit($limit)
            ->get('blog_posts');
    }
}
```

## 43.4. Фронтенд-контроллер (frontend.php)

```php
<?php
/**
 * frontend.php
 * Фронтенд-контроллер блога
 */

class blog extends cmsFrontend {
    
    /**
     * Действие по умолчанию - список постов
     */
    public function actionIndex() {
        // Получение параметров
        $page = $this->request->get('page', 1);
        $category_id = $this->request->get('category', 0);
        $tag = $this->request->get('tag', '');
        
        // Подготовка фильтра
        $filter = [
            'page' => $page,
            'per_page' => 10,
            'category_id' => $category_id,
            'tag' => $tag
        ];
        
        // Получение постов
        $posts = $this->model->getPosts($filter);
        
        // Получение категорий
        $categories = $this->model->getCategories();
        
        // Получение популярных постов для сайдбара
        $popular_posts = $this->model->getPopularPosts(5);
        
        // Подготовка данных для шаблона
        $data = [
            'posts' => $posts,
            'categories' => $categories,
            'popular_posts' => $popular_posts,
            'current_category' => $category_id ? $this->model->getItemById('blog_categories', $category_id) : null,
            'current_tag' => $tag,
            'page' => $page,
            'total_pages' => ceil($this->model->getItemsCount('blog_posts') / 10)
        ];
        
        // Установка заголовка страницы
        $this->setPageTitle('Блог');
        
        // Регистрация стилей компонента
        $this->cms_template->addControllerCSS('styles.css', 'blog');
        
        // Рендеринг шаблона
        return $this->render('index', $data);
    }
    
    /**
     * Просмотр отдельного поста
     * @param int $id ID поста
     * @param string $slug Слаг поста (для ЧПУ)
     */
    public function actionView($id, $slug = '') {
        // Получение поста
        $post = $this->model->getPost($id);
        
        if (!$post) {
            return $this->error404();
        }
        
        // Проверка прав доступа
        if (!$post['is_published'] && !$this->cms_user->is_admin) {
            return $this->error403('Пост не опубликован');
        }
        
        // Установка SEO-метаданных
        $this->setPageTitle($post['title']);
        $this->setPageDescription($post['excerpt'] ?: string_short(strip_tags($post['content']), 160));
        $this->setPageKeywords(implode(', ', $post['tags_array']));
        
        // Получение связанных постов
        $related_posts = $this->model->getPosts([
            'category_id' => $post['category_id'],
            'show_all' => true,
            'limit' => 3
        ]);
        
        // Исключение текущего поста из связанных
        $related_posts = array_filter($related_posts, function($p) use ($id) {
            return $p['id'] != $id;
        });
        
        // Подготовка данных для шаблона
        $data = [
            'post' => $post,
            'related_posts' => array_values($related_posts),
            'can_edit' => $this->cms_user->is_admin || 
                         ($this->cms_user->id == $post['user_id'] && $this->cms_user->isAllowed('blog', 'edit_own'))
        ];
        
        // Регистрация стилей
        $this->cms_template->addControllerCSS('styles.css', 'blog');
        
        // Рендеринг шаблона
        return $this->render('view', $data);
    }
    
    /**
     * Список постов по категории
     * @param string $slug Слаг категории
     */
    public function actionCategory($slug) {
        // Получение категории
        $category = $this->model->getCategoryBySlug($slug);
        
        if (!$category) {
            return $this->error404();
        }
        
        // Получение параметров
        $page = $this->request->get('page', 1);
        
        // Получение постов категории
        $posts = $this->model->getPosts([
            'page' => $page,
            'per_page' => 10,
            'category_id' => $category['id']
        ]);
        
        // Подготовка данных для шаблона
        $data = [
            'posts' => $posts,
            'category' => $category,
            'page' => $page,
            'total_pages' => ceil($this->model->getItemsCount('blog_posts', [
                'category_id' => $category['id']
            ]) / 10)
        ];
        
        // Установка заголовка
        $this->setPageTitle('Блог - ' . $category['title']);
        
        // Рендеринг шаблона
        return $this->render('category', $data);
    }
    
    /**
     * Создание нового поста (требует авторизации)
     */
    public function actionAdd() {
        // Проверка авторизации
        if (!$this->cms_user->is_logged) {
            return $this->redirectToLogin();
        }
        
        // Проверка прав
        if (!$this->cms_user->isAllowed('blog', 'add')) {
            return $this->error403('У вас нет прав на создание постов');
        }
        
        // Обработка формы
        if ($this->request->isPost()) {
            return $this->handleAddForm();
        }
        
        // Получение категорий для формы
        $categories = $this->model->getCategories();
        
        // Подготовка данных для шаблона
        $data = [
            'categories' => $categories,
            'post' => null,
            'form_action' => href_to('blog', 'add')
        ];
        
        // Установка заголовка
        $this->setPageTitle('Создать пост');
        
        // Рендеринг формы
        return $this->render('form', $data);
    }
    
    /**
     * Редактирование поста
     * @param int $id ID поста
     */
    public function actionEdit($id) {
        // Проверка авторизации
        if (!$this->cms_user->is_logged) {
            return $this->redirectToLogin();
        }
        
        // Получение поста
        $post = $this->model->getItemById('blog_posts', $id);
        
        if (!$post) {
            return $this->error404();
        }
        
        // Проверка прав
        $can_edit = $this->cms_user->is_admin || 
                   ($this->cms_user->id == $post['user_id'] && $this->cms_user->isAllowed('blog', 'edit_own')) ||
                   $this->cms_user->isAllowed('blog', 'edit');
        
        if (!$can_edit) {
            return $this->error403('У вас нет прав на редактирование этого поста');
        }
        
        // Обработка формы
        if ($this->request->isPost()) {
            return $this->handleEditForm($id);
        }
        
        // Получение категорий для формы
        $categories = $this->model->getCategories();
        
        // Подготовка данных для шаблона
        $data = [
            'categories' => $categories,
            'post' => $post,
            'form_action' => href_to('blog', 'edit', $id)
        ];
        
        // Установка заголовка
        $this->setPageTitle('Редактировать пост - ' . $post['title']);
        
        // Рендеринг формы
        return $this->render('form', $data);
    }
    
    /**
     * Обработка формы создания поста
     */
    private function handleAddForm() {
        // Валидация данных
        $data = $this->validatePostData();
        
        if (!$data) {
            return $this->error('Ошибка валидации данных');
        }
        
        // Сохранение поста
        $post_id = $this->model->savePost($data);
        
        // Редирект на страницу поста
        $this->redirectTo(href_to('blog', 'view', [$post_id, $data['slug']]));
    }
    
    /**
     * Обработка формы редактирования поста
     * @param int $id ID поста
     */
    private function handleEditForm($id) {
        // Валидация данных
        $data = $this->validatePostData();
        
        if (!$data) {
            return $this->error('Ошибка валидации данных');
        }
        
        // Сохранение поста
        $this->model->savePost($data, $id);
        
        // Редирект на страницу поста
        $this->redirectTo(href_to('blog', 'view', [$id, $data['slug']]));
    }
    
    /**
     * Валидация данных поста
     * @return array|false
     */
    private function validatePostData() {
        // Получение данных из запроса
        $title = trim($this->request->post('title', ''));
        $content = $this->request->post('content', '');
        $excerpt = trim($this->request->post('excerpt', ''));
        $category_id = (int)$this->request->post('category_id', 0);
        $tags = trim($this->request->post('tags', ''));
        $is_published = (bool)$this->request->post('is_published', 0);
        
        // Валидация заголовка
        if (mb_strlen($title) < 3) {
            $this->cms_template->addNotice('Заголовок должен содержать минимум 3 символа', 'error');
            return false;
        }
        
        // Валидация контента
        if (mb_strlen(strip_tags($content)) < 10) {
            $this->cms_template->addNotice('Контент должен содержать минимум 10 символов', 'error');
            return false;
        }
        
        // Валидация категории
        if ($category_id && !$this->model->getItemById('blog_categories', $category_id)) {
            $this->cms_template->addNotice('Выбранная категория не существует', 'error');
            return false;
        }
        
        // Формирование данных
        $data = [
            'user_id' => $this->cms_user->id,
            'category_id' => $category_id,
            'title' => $title,
            'content' => $content,
            'excerpt' => $excerpt ?: string_short(strip_tags($content), 200),
            'tags' => $tags,
            'is_published' => $is_published
        ];
        
        // Обработка изображения
        if (!empty($_FILES['image']['name'])) {
            $image_path = $this->uploadImage($_FILES['image']);
            if ($image_path) {
                $data['image'] = $image_path;
            }
        }
        
        return $data;
    }
    
    /**
     * Загрузка изображения
     * @param array $file Данные файла
     * @return string|false
     */
    private function uploadImage($file) {
        // Проверка типа файла
        $allowed_extensions = ['jpg', 'jpeg', 'png', 'gif'];
        $extension = mb_strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        
        if (!in_array($extension, $allowed_extensions)) {
            $this->cms_template->addNotice('Недопустимый формат изображения', 'error');
            return false;
        }
        
        // Проверка размера
        $max_size = 5 * 1024 * 1024; // 5MB
        if ($file['size'] > $max_size) {
            $this->cms_template->addNotice('Размер изображения не должен превышать 5 МБ', 'error');
            return false;
        }
        
        // Генерация имени файла
        $filename = 'blog_' . uniqid() . '.' . $extension;
        $upload_dir = PATH . '/uploads/blog/';
        
        // Создание директории, если не существует
        if (!is_dir($upload_dir)) {
            mkdir($upload_dir, 0755, true);
        }
        
        // Перемещение файла
        $destination = $upload_dir . $filename;
        
        if (move_uploaded_file($file['tmp_name'], $destination)) {
            return '/uploads/blog/' . $filename;
        }
        
        return false;
    }
```

## 43.6. Модель бэкенда (backend/model.php)

```php
<?php
/**
 * backend/model.php
 * Модель бэкенда блога
 */

class modelBackendBlog extends modelBlog {
    
    public $perpage = 20;
    
    /**
     * Получение списка постов для админки
     * @param int $page Номер страницы
     * @param array $filter Фильтры
     * @return array
     */
    public function getPostsList($page = 1, $filter = []) {
        $this->select('blog_posts', 'p');
        $this->select('users', 'u', 'u.id = p.user_id');
        
        // Фильтрация по статусу
        if (!empty($filter['status'])) {
            switch ($filter['status']) {
                case 'published':
                    $this->filterEqual('p.is_published', 1);
                    $this->filterEqual('p.is_deleted', 0);
                    break;
                case 'draft':
                    $this->filterEqual('p.is_published', 0);
                    $this->filterEqual('p.is_deleted', 0);
                    break;
                case 'deleted':
                    $this->filterEqual('p.is_deleted', 1);
                    break;
            }
        }
        
        // Фильтрация по автору
        if (!empty($filter['user_id'])) {
            $this->filterEqual('p.user_id', $filter['user_id']);
        }
        
        // Фильтрация по категории
        if (!empty($filter['category_id'])) {
            $this->filterEqual('p.category_id', $filter['category_id']);
        }
        
        // Поиск по заголовку
        if (!empty($filter['search'])) {
            $this->filterLike('p.title', '%' . $filter['search'] . '%');
        }
        
        // Сортировка
        $this->orderBy('p.date_pub', 'DESC');
        
        // Пагинация
        $this->limitPage($page, $this->perpage);
        
        // Получение постов
        $posts = $this->get();
        
        // Обогащение данными
        foreach ($posts as &$post) {
            $post = $this->preparePost($post);
        }
        
        return $posts;
    }
    
    /**
     * Получение количества постов с фильтрами
     * @param array $filter Фильтры
     * @return int
     */
    public function getPostsCount($filter = []) {
        $this->selectOnlyCount('blog_posts', 'p');
        
        // Фильтрация (аналогично getPostsList)
        if (!empty($filter['status'])) {
            switch ($filter['status']) {
                case 'published':
                    $this->filterEqual('p.is_published', 1);
                    $this->filterEqual('p.is_deleted', 0);
                    break;
                case 'draft':
                    $this->filterEqual('p.is_published', 0);
                    $this->filterEqual('p.is_deleted', 0);
                    break;
                case 'deleted':
                    $this->filterEqual('p.is_deleted', 1);
                    break;
            }
        }
        
        if (!empty($filter['user_id'])) {
            $this->filterEqual('user_id', $filter['user_id']);
        }
        
        if (!empty($filter['category_id'])) {
            $this->filterEqual('category_id', $filter['category_id']);
        }
        
        if (!empty($filter['search'])) {
            $this->filterLike('title', '%' . $filter['search'] . '%');
        }
        
        return $this->getCount();
    }
    
    /**
     * Получение дерева категорий
     * @return array
     */
    public function getCategoriesTree() {
        $categories = $this->getCategories();
        return $this->buildTree($categories);
    }
    
    /**
     * Построение дерева категорий
     * @param array $categories Список категорий
     * @param int $parent_id ID родительской категории
     * @return array
     */
    private function buildTree($categories, $parent_id = 0) {
        $tree = [];
        
        foreach ($categories as $category) {
            if ($category['parent_id'] == $parent_id) {
                $category['children'] = $this->buildTree($categories, $category['id']);
                $tree[] = $category;
            }
        }
        
        return $tree;
    }
}
```

## 43.11. Обработчики событий (hooks.php)

```php
<?php
/**
 * hooks.php
 * Обработчики событий блога
 */

class blog extends cmsAction {
    
    /**
     * Хук: после добавления поста
     * @param array $post Данные поста
     */
    public function hookBlogAfterAdd($post) {
        // Логирование
        $this->logActivity('blog_post_created', [
            'post_id' => $post['id'],
            'post_title' => $post['title'],
            'user_id' => $post['user_id']
        ]);
        
        // Уведомление подписчиков (если включено)
        if ($this->getOption('notify_subscribers', false)) {
            $this->notifySubscribers($post);
        }
        
        return $post;
    }
    
    /**
     * Хук: после обновления поста
     * @param array $post Данные поста
     * @param array $old_post Старые данные поста
     */
    public function hookBlogAfterUpdate($post, $old_post) {
        // Логирование изменений
        $changes = [];
        
        if ($post['title'] != $old_post['title']) {
            $changes[] = 'title';
        }
        
        if ($post['content'] != $old_post['content']) {
            $changes[] = 'content';
        }
        
        if (!empty($changes)) {
            $this->logActivity('blog_post_updated', [
                'post_id' => $post['id'],
                'changes' => $changes,
                'user_id' => $this->cms_user->id
            ]);
        }
        
        return $post;
    }
    
    /**
     * Хук: после удаления поста
     * @param array $post Данные поста
     */
    public function hookBlogAfterDelete($post) {
        // Логирование
        $this->logActivity('blog_post_deleted', [
            'post_id' => $post['id'],
            'post_title' => $post['title'],
            'user_id' => $this->cms_user->id
        ]);
        
        // Удаление связанного контента (комментарии, лайки и т.д.)
        $this->cleanupRelatedContent($post['id']);
        
        return $post;
    }
    
    /**
     * Хук: интеграция с компонентом поиска
     * @param array $indexable_items Массив элементов для индексации
     * @return array
     */
    public function hookSearchGetIndexableItems($indexable_items) {
        $model = cmsCore::getModel('blog');
        $posts = $model->getPosts(['show_all' => true]);
        
        foreach ($posts as $post) {
            $indexable_items[] = [
                'component' => 'blog',
                'item_id' => $post['id'],
                'title' => $post['title'],
                'content' => $post['content'],
                'url' => href_to('blog', 'view', [$post['id'], $post['slug']]),
                'date_pub' => $post['date_pub']
            ];
        }
        
        return $indexable_items;
    }
    
    /**
     * Хук: интеграция с компонентом активности
     * @param array $activity_types Типы активности
     * @return array
     */
    public function hookActivityGetTypes($activity_types) {
        $activity_types['blog_post_created'] = [
            'title' => 'Создал(а) пост в блоге',
            'component' => 'blog'
        ];
        
        $activity_types['blog_post_updated'] = [
            'title' => 'Обновил(а) пост в блоге',
            'component' => 'blog'
        ];
        
        return $activity_types;
    }
    
    /**
     * Логирование активности
     */
    private function logActivity($type, $data) {
        if (cmsCore::isControllerExists('activity')) {
            $activity = cmsCore::getController('activity');
            $activity->addEntry([
                'user_id' => $data['user_id'],
                'type' => $type,
                'component' => 'blog',
                'item_id' => $data['post_id'] ?? 0,
                'data' => $data
            ]);
        }
    }
    
    /**
     * Уведомление подписчиков
     */
    private function notifySubscribers($post) {
        // Получение подписчиков
        $subscribers = $this->getSubscribers();
        
        foreach ($subscribers as $subscriber) {
            $this->sendNotification($subscriber, $post);
        }
    }
    
    /**
     * Очистка связанного контента
     */
    private function cleanupRelatedContent($post_id) {
        // Удаление комментариев (если компонент comments существует)
        if (cmsCore::isControllerExists('comments')) {
            $comments_model = cmsCore::getModel('comments');
            $comments_model->deleteTargetComments('blog', $post_id);
        }
        
        // Удаление лайков (если компонент ratings существует)
        if (cmsCore::isControllerExists('ratings')) {
            $ratings_model = cmsCore::getModel('ratings');
            $ratings_model->deleteTargetRatings('blog', $post_id);
        }
    }
    
    /**
     * Получение подписчиков
     */
    private function getSubscribers() {
        // Логика получения подписчиков
        // Например, из таблицы подписок или через компонент subscriptions
        return [];
    }
    
    /**
     * Отправка уведомления
     */
    private function sendNotification($subscriber, $post) {
        // Логика отправки уведомления
        // Например, через email или через компонент notifications
    }
}
```
